<script>
    function addResultTemplate() {
        const container = document.getElementById('result-templates');
        const template = `
        <div class="result-template-group flex gap-4 items-start">
            <div class="flex-1 space-y-2">
                <input type="text" name="result_text[]" 
                       placeholder="Display text" 
                       class="w-full px-4 py-2 rounded-lg border-0 ring-1 ring-gray-200 dark:ring-gray-700 bg-white dark:bg-gray-900">
                <textarea name="result_value[]" 
                          rows="2"
                          placeholder="Full result text (supports Markdown)"
                          class="w-full px-4 py-2 rounded-lg border-0 ring-1 ring-gray-200 dark:ring-gray-700 bg-white dark:bg-gray-900"></textarea>
            </div>
            <button type="button" onclick="removeResultTemplate(this)" 
                    class="mt-2 px-2 py-1 text-red-500 hover:text-red-600">
                <i class="fas fa-trash"></i>
            </button>
        </div>
    `;
        container.insertAdjacentHTML('beforeend', template);
    }

    function removeResultTemplate(button) {
        button.closest('.result-template-group').remove();
    }
</script>

@lab_web_bp.route('/lab/tests/create', methods=['GET', 'POST'])
@login_required
def create_test_type():
    """Create new lab test type"""
    try:
        if request.method == 'POST':
            # Validate required fields
            required_fields = ['name', 'price', 'description']
            if not all(request.form.get(field) for field in required_fields):
                flash('Please fill all required fields', 'danger')
                return redirect(url_for('lab_web.create_test_type'))
            
            new_test = Service(
                name=request.form['name'],
                category='Laboratory',
                price=float(request.form['price']),
                description=request.form['description'],
                duration=int(request.form.get('duration', 0)),
                predefined_results=request.form.get('predefined_results', '').split(','),
                storage_conditions=request.form.get('storage_conditions'),
                requires_prescription='requires_prescription' in request.form
            )
            
            db.session.add(new_test)
            db.session.commit()
            
            flash('Lab test type created successfully', 'success')
            return redirect(url_for('lab_web.manage_test_types'))
        
        return render_template(
            'edit_test_type.html',
            title="Create New Lab Test Type",
            test_type=None
        )

    
    except Exception as e:
        db.session.rollback()
        flash(f'Error creating test type: {str(e)}', 'danger')
        return redirect(url_for('lab_web.manage_test_types'))
